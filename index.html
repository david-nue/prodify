How to connect
1. Create a free project at supabase.com
2. Run the SQL below in your SQL editor
3. Paste your URL + anon key into the file
SQL to run
create table users (
  username text primary key,
  pass_hash text,
  display_name text,
  tasks text default '[]',
  journal text default '[]',
  subjects text default '[]',
  cal_evs text default '[]',
  widgets text default '[]',
  notes text default '{}',
  prefs text default '{}',
  joined_at timestamptz default now()
);
alter table users enable row level security;
create policy "public" on users for all using (true);
